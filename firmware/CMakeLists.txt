cmake_minimum_required(VERSION 3.16)
set(MINIDASH_BOARD "esp32_s3" CACHE STRING "Hardware target: esp32_s3 or minidash_cyd")
if(MINIDASH_BOARD STREQUAL "minidash_cyd")
    set(PROJECT_VER "cyd-integration-dev")
    set(IDF_TARGET esp32)
    option(MINIDASH_CYD_LISTEN_ONLY "Build a CYD image with diagnostic TX disabled" OFF)
    if(MINIDASH_CYD_LISTEN_ONLY)
        set(SDKCONFIG "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.cyd-listen.generated")
        set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.cyd;${CMAKE_CURRENT_LIST_DIR}/sdkconfig.cyd.listen-only")
    else()
        set(SDKCONFIG "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.cyd.generated")
        set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.cyd")
    endif()
elseif(NOT MINIDASH_BOARD STREQUAL "esp32_s3")
    message(FATAL_ERROR "Unknown MINIDASH_BOARD=${MINIDASH_BOARD}")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(MINIDASH_BOARD STREQUAL "minidash_cyd")
    idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_CURRENT_LIST_DIR}/dependencies.cyd.lock")
endif()
project(vag_minidash)
