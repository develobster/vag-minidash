cmake_minimum_required(VERSION 3.16)
project(minidash_core C)
set(CMAKE_C_STANDARD 11)
# Same physical sources used by ESP-IDF; no duplicated target source trees.
set(MD_CORE_DIR ${CMAKE_CURRENT_LIST_DIR}/../firmware/main)
add_library(minidash-core STATIC
    ${MD_CORE_DIR}/can_pipeline.c
    ${MD_CORE_DIR}/generated_can_signals.c
    ${MD_CORE_DIR}/vag_can_db.c ${MD_CORE_DIR}/drag_timer.c
    ${MD_CORE_DIR}/ui_core.c ${MD_CORE_DIR}/ui_theme.c
    ${MD_CORE_DIR}/ui_pixel_art.c
    ${MD_CORE_DIR}/ui_startup.c ${MD_CORE_DIR}/ui_i18n.c ${MD_CORE_DIR}/ui_model.c
    ${MD_CORE_DIR}/ui_platform.c ${MD_CORE_DIR}/ui_configuration.c
    ${MD_CORE_DIR}/ui_touch_map.c ${MD_CORE_DIR}/display_serial_protocol.c
    ${MD_CORE_DIR}/diagnostic_transport.c ${MD_CORE_DIR}/acquisition.c ${MD_CORE_DIR}/ui_instrument_profiles.c
    ${MD_CORE_DIR}/window_statistics.c ${MD_CORE_DIR}/performance_timer.c ${MD_CORE_DIR}/derived_engine.c)
target_include_directories(minidash-core PUBLIC ${MD_CORE_DIR})
set_target_properties(minidash-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(minidash-vag-native SHARED vag_native_api.c)
target_link_libraries(minidash-vag-native PRIVATE minidash-core)
if(NOT MSVC)
    target_compile_options(minidash-core PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(minidash-core PUBLIC m)
endif()

include(CTest)
set(MD_LVGL_SOURCE_DIR "" CACHE PATH "Optional LVGL 9.5 source for actual software-render acceptance")
if(MD_LVGL_SOURCE_DIR)
    set(LV_BUILD_CONF_PATH ${CMAKE_CURRENT_LIST_DIR}/lvgl_stubs/lv_conf.h CACHE PATH "" FORCE)
    set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
    set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
    add_subdirectory(${MD_LVGL_SOURCE_DIR} ${CMAKE_BINARY_DIR}/lvgl)
    add_executable(minidash-auto-ui-tests test_automotive_screens.c ${MD_CORE_DIR}/automotive_screens.c)
    target_link_libraries(minidash-auto-ui-tests PRIVATE minidash-core lvgl::lvgl)
    add_test(NAME automotive-lvgl-render COMMAND minidash-auto-ui-tests ${CMAKE_BINARY_DIR}/ui-previews)
endif()
if(BUILD_TESTING)
    # Exercise CYD's source-mode policy independently of the driver.
    add_executable(minidash-cyd-mode-tests test_cyd_mode.c ${MD_CORE_DIR}/cyd_mode.c)
    target_include_directories(minidash-cyd-mode-tests PRIVATE ${MD_CORE_DIR})
    add_test(NAME cyd-mode-policy COMMAND minidash-cyd-mode-tests)
    # Exercise simulation and portal contracts independently of firmware I/O.
    add_executable(minidash-cyd-reference-tests test_cyd.c ${MD_CORE_DIR}/simulation.c ${MD_CORE_DIR}/captive_dns.c)
    target_include_directories(minidash-cyd-reference-tests PRIVATE ${MD_CORE_DIR})
    if(NOT MSVC)
        target_link_libraries(minidash-cyd-reference-tests PRIVATE m)
    endif()
    add_test(NAME cyd-simulation-and-portal COMMAND minidash-cyd-reference-tests)
    add_executable(minidash-vag-nvs-tests test_vag_nvs.c ${MD_CORE_DIR}/vag_can_db_nvs.c)
    target_include_directories(minidash-vag-nvs-tests PRIVATE nvs_stubs)
    target_link_libraries(minidash-vag-nvs-tests PRIVATE minidash-core)
    add_test(NAME vag-config-nvs COMMAND minidash-vag-nvs-tests)
    add_executable(minidash-vag-db-tests test_vag_can_db.c)
    target_link_libraries(minidash-vag-db-tests PRIVATE minidash-core)
    add_test(NAME vag-can-database COMMAND minidash-vag-db-tests)
    add_executable(minidash-drag-tests test_drag_timer.c)
    target_link_libraries(minidash-drag-tests PRIVATE minidash-core)
    add_test(NAME drag-timer COMMAND minidash-drag-tests)
    foreach(power_variant rx ignition timer disabled)
        add_executable(minidash-power-${power_variant}-tests test_power.c ${MD_CORE_DIR}/power_manager.c)
        target_include_directories(minidash-power-${power_variant}-tests PRIVATE power_stubs)
        target_link_libraries(minidash-power-${power_variant}-tests PRIVATE minidash-core)
        if(power_variant STREQUAL ignition)
            target_compile_definitions(minidash-power-${power_variant}-tests PRIVATE MD_TEST_WAKE_GPIO=6 MD_TEST_WAKE_LEVEL=1)
        elseif(power_variant STREQUAL timer)
            target_compile_definitions(minidash-power-${power_variant}-tests PRIVATE MD_TEST_WAKE_GPIO=-1)
        elseif(power_variant STREQUAL disabled)
            target_compile_definitions(minidash-power-${power_variant}-tests PRIVATE MD_TEST_SLEEP=0)
        endif()
        add_test(NAME power-${power_variant} COMMAND minidash-power-${power_variant}-tests)
    endforeach()
    find_package(Threads REQUIRED)
    add_executable(minidash-pipeline-tests test_can_pipeline.c)
    target_link_libraries(minidash-pipeline-tests PRIVATE minidash-core Threads::Threads)
    add_test(NAME can-realtime-pipeline COMMAND minidash-pipeline-tests)
    add_executable(minidash-transport-tests test_diagnostic_transport.c)
    target_link_libraries(minidash-transport-tests PRIVATE minidash-core)
    add_test(NAME diagnostic-transport COMMAND minidash-transport-tests)
    add_executable(minidash-can-link-tests test_can_link.c ${MD_CORE_DIR}/can_link_tx.c)
    target_include_directories(minidash-can-link-tests PRIVATE can_stubs ${MD_CORE_DIR})
    add_test(NAME can-transmit-confirmation COMMAND minidash-can-link-tests)
    # Exercise the real acquisition/UI sources with the CYD memory limits too.
    get_target_property(MD_CORE_SOURCES minidash-core SOURCES)
    add_library(minidash-core-cyd STATIC ${MD_CORE_SOURCES})
    target_include_directories(minidash-core-cyd PUBLIC ${MD_CORE_DIR})
    target_compile_definitions(minidash-core-cyd PUBLIC CONFIG_MINIDASH_CYD=1)
    if(NOT MSVC)
        target_compile_options(minidash-core-cyd PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(minidash-core-cyd PUBLIC m)
    endif()
    if(MD_LVGL_SOURCE_DIR)
        add_executable(minidash-cyd-auto-ui-tests test_automotive_screens.c ${MD_CORE_DIR}/automotive_screens.c)
        target_link_libraries(minidash-cyd-auto-ui-tests PRIVATE minidash-core-cyd lvgl::lvgl)
        add_test(NAME cyd-automotive-lvgl-render COMMAND minidash-cyd-auto-ui-tests ${CMAKE_BINARY_DIR}/cyd-ui-previews)
    endif()
    add_executable(minidash-cyd-acquisition-tests test_acquisition.c)
    target_link_libraries(minidash-cyd-acquisition-tests PRIVATE minidash-core-cyd)
    add_test(NAME cyd-acquisition COMMAND minidash-cyd-acquisition-tests)
    add_executable(minidash-cyd-core-tests tests.c ${MD_CORE_DIR}/ui_platform_selftest.c)
    target_link_libraries(minidash-cyd-core-tests PRIVATE minidash-core-cyd)
    add_test(NAME cyd-portable-core COMMAND minidash-cyd-core-tests)
    add_executable(minidash-pixel-tests test_pixel_art.c)
    target_link_libraries(minidash-pixel-tests PRIVATE minidash-core)
    add_test(NAME pixel-art-and-transitions COMMAND minidash-pixel-tests)
    foreach(BOARD cyd s3)
        add_executable(minidash-orientation-${BOARD} test_orientation.c ${MD_CORE_DIR}/display_manager.c)
        target_include_directories(minidash-orientation-${BOARD} PRIVATE display_stubs ${MD_CORE_DIR})
        if(BOARD STREQUAL "cyd")
            target_compile_definitions(minidash-orientation-${BOARD} PRIVATE CONFIG_MINIDASH_CYD=1)
        endif()
        add_test(NAME orientation-${BOARD} COMMAND minidash-orientation-${BOARD})
    endforeach()
    add_executable(minidash-analytics-pipeline-tests test_analytics_pipeline.c)
    target_link_libraries(minidash-analytics-pipeline-tests PRIVATE minidash-core)
    add_test(NAME analytics-acquisition-pipeline COMMAND minidash-analytics-pipeline-tests)
    add_executable(minidash-analytics-harness analytics_harness.c)
    target_link_libraries(minidash-analytics-harness PRIVATE minidash-core)
    add_executable(minidash-analytics-tests test_analytics.c)
    target_link_libraries(minidash-analytics-tests PRIVATE minidash-core)
    if(UNIX AND NOT APPLE)
        target_compile_definitions(minidash-analytics-tests PRIVATE MD_TEST_ALLOCATIONS=1)
        target_link_options(minidash-analytics-tests PRIVATE -Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free)
    endif()
    add_test(NAME derived-analytics COMMAND minidash-analytics-tests)
    add_executable(minidash-acquisition-harness acquisition_harness.c)
    target_link_libraries(minidash-acquisition-harness PRIVATE minidash-core)
    add_executable(minidash-acquisition-tests test_acquisition.c)
    target_link_libraries(minidash-acquisition-tests PRIVATE minidash-core)
    add_test(NAME standalone-acquisition COMMAND minidash-acquisition-tests)
    add_executable(minidash-instrument-tests test_instruments.c)
    target_link_libraries(minidash-instrument-tests PRIVATE minidash-core)
    add_test(NAME instrument-profile-contract COMMAND minidash-instrument-tests)
    add_executable(minidash-core-tests tests.c ${MD_CORE_DIR}/ui_platform_selftest.c)
    target_link_libraries(minidash-core-tests PRIVATE minidash-core)
    add_test(NAME portable-core COMMAND minidash-core-tests)
    add_executable(minidash-backlight-tests test_backlight.c ${MD_CORE_DIR}/backlight_controller.c)
    target_include_directories(minidash-backlight-tests PRIVATE backlight_stubs ${MD_CORE_DIR})
    add_test(NAME backlight-driver-contract COMMAND minidash-backlight-tests)
    set(MD_JSON_SOURCE_DIR "" CACHE PATH "Optional cJSON source directory for editor contract tests")
    if(MD_JSON_SOURCE_DIR)
        add_executable(minidash-validation-tests test_validation.c
            ${MD_CORE_DIR}/validation_json.c ${MD_JSON_SOURCE_DIR}/cJSON.c)
        target_include_directories(minidash-validation-tests PRIVATE ${MD_JSON_SOURCE_DIR})
        target_link_libraries(minidash-validation-tests PRIVATE minidash-core)
        add_test(NAME car-validation COMMAND minidash-validation-tests)
        add_executable(minidash-json-tests test_json.c
            ${MD_CORE_DIR}/ui_configuration_json.c ${MD_JSON_SOURCE_DIR}/cJSON.c)
        target_include_directories(minidash-json-tests PRIVATE ${MD_JSON_SOURCE_DIR})
        target_link_libraries(minidash-json-tests PRIVATE minidash-core)
        add_test(NAME configuration-json COMMAND minidash-json-tests)
    endif()
endif()
